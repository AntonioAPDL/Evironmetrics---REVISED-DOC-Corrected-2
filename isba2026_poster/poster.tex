\SectionHead{Question}{Fast blending of corrected products}

\begin{KeyQuestion}
{\SmallBodyFont\bfseries\color{TitleSlate}
How can large retrospective product archives be used efficiently at an
operational forecast origin to learn bias for each source, calibrate forecast
ensembles across lead times, and synthesize predictive quantiles for the target
process?\par}
\end{KeyQuestion}

\vspace{0.27cm}
{\SmallBodyFont
Case study: daily San Lorenzo River flow at the USGS Big Trees gauge. The
framework learns dynamic corrections from USGS observations and retrospective
ECMWF/GloFAS and NOAA/NWS products, then carries those corrections into the
forecast window to synthesize calibrated predictive flow quantiles. This is
needed because product families differ in lead time, update cycle, ensemble
structure, and uncertainty representation.\par}

\SectionHead{Model}{One source-aware dynamic quantile system}
\PosterGraphic[18.4cm]{model_schematic.pdf}{Dynamic quantile synthesis schematic}
\FigureCaption[From heterogeneous inputs to one distribution.]{The dynamic
state includes trend and seasonality, source-specific discrepancy, and retained
exogenous-covariate effects. Quantile-specific posterior forecasts are
synthesized into one predictive distribution.}

\vspace{0.34cm}
\begin{SoftNote}
  {\SmallBodyFont\bfseries\color{TitleSlate}Expanded quantile state-space model\par}
  \vspace{0.08cm}
  {\CaptionFont\color{MutedSlate}
  For each target quantile \(p_0\) and forecast origin \(T\), observed flow,
  retrospective product histories, and issued forecast members share one latent
  river-flow quantile plus source-specific corrections.\par}
  \vspace{-0.20cm}
  {\ModelEqFont\color{PosterInk}
  \setlength{\jot}{0.045cm}
  \begin{alignat*}{3}
  \text{Observed measurements:}\quad&
  y_t^o
  &\sim{}& \operatorname{exAL}_{p_0}\!\left(\mathbf{F}_t' \boldsymbol{\theta}_t + \zeta_t, \sigma^o, \gamma^o\right),
  &\qquad& 1 \le t \le T,
  \\
  \text{Retrospective products:}\quad&
  z_t^j
  &\sim{}& \operatorname{exAL}_{p_0}\!\left(\mathbf{F}_t'(\boldsymbol{\theta}_t + \boldsymbol{\delta}_t^j) + \zeta_t, \sigma^j, \gamma^j\right),
  &\qquad& j \in \mathcal{J},\ 1 \le t \le T,
  \\
  \text{Forecast products:}\quad&
  y_T^{j,i}(k)
  &\sim{}& \operatorname{exAL}_{p_0}\!\left(\mathbf{F}_{T+k}'(\boldsymbol{\theta}_{T+k} + \boldsymbol{\delta}_{T+k}^j) + \zeta_{T+k}, \sigma^j, \gamma^j\right),
  &\qquad& j \in \mathcal{J},\ i \in \mathcal{I}_j,\ k \in \mathcal{K}_j,
  \\
  \text{Trend and seasonal states:}\quad&
  \boldsymbol{\theta}_t \mid \boldsymbol{\theta}_{t-1}
  &\sim{}& \mathcal{N}\!\left(\mathbf{G}_t \boldsymbol{\theta}_{t-1}, \mathbf{W}_t^\theta\right),
  &\qquad& 1 \le t \le T+K_{\max},
  \\
  \text{Discrepancy states:}\quad&
  \boldsymbol{\delta}_t^j \mid \boldsymbol{\delta}_{t-1}^j
  &\sim{}& \mathcal{N}\!\left(\mathbf{G}_t \boldsymbol{\delta}_{t-1}^j, \mathbf{W}_t^{\delta^j}\right),
  &\qquad& j \in \mathcal{J},\ 1 \le t \le T+K_{\max},
  \\
  \text{Transfer adjustment:}\quad&
  \zeta_t \mid \zeta_{t-1}, \boldsymbol{\psi}_{t-1}
  &\sim{}& \mathcal{N}\!\left(\lambda \zeta_{t-1} + \mathbf{x}_t'\boldsymbol{\psi}_{t-1}, w_t^\zeta\right),
  &\qquad& 1 \le t \le T+K_{\max},
  \\
  \text{Covariate-effect states:}\quad&
  \boldsymbol{\psi}_t \mid \boldsymbol{\psi}_{t-1}
  &\sim{}& \mathcal{N}\!\left(\boldsymbol{\psi}_{t-1}, \mathbf{W}_t^\psi\right),
  &\qquad& 1 \le t \le T+K_{\max}.
  \end{alignat*}
  }
  \vspace{-0.18cm}
  {\CaptionFont\color{MutedSlate}
  Here exAL is the extended asymmetric Laplace likelihood; \(\sigma\) and
  \(\gamma\) are source-specific scale and skewness parameters.
  \(\boldsymbol{\theta}_t\) is the latent river-flow quantile,
  \(\boldsymbol{\delta}_t^j\) corrects product family \(j\), and
  \((\zeta_t,\boldsymbol{\psi}_t)\) carries exogenous covariate effects.\par}
\end{SoftNote}

\SectionHead{Interpretation}{What this experiment supports}
\begin{PosterItems}
  \item Retrospective histories estimate source-specific corrections before the
  forecast window.
  \item Forecast ensembles and covariates enter the same dynamic quantile
  system, instead of being averaged after the fact.
  \item Observation-only AL/exAL quantile references have roughly 5.4--39x
  higher 28-day CRPS across these origins, highlighting the value of external
  products and exogenous covariates.
  \item The five-window evaluation is a targeted stress test; broader regime
  coverage remains the next validation step.
\end{PosterItems}
