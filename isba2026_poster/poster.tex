\SectionHead{Question}{Why heterogeneous forecasts are difficult}

\begin{KeyQuestion}
{\SmallBodyFont\bfseries\color{TitleSlate}
How can hydrologic products with different horizons, discrepancies, and
uncertainty representations be reconciled into calibrated predictive quantiles?\par}
\end{KeyQuestion}

\vspace{0.27cm}
{\SmallBodyFont
The target is daily San Lorenzo River flow at the Big Trees gauge. The model
synthesizes observations, operational and retrospective forecast products, and
hydrometeorological transfer information without treating the sources as
exchangeable.\par}

\vspace{0.24cm}
\begin{SoftNote}
\SourceRow{USGSInk}{USGS observations}{state updating and post-cutoff verification.}
\SourceRow{GloFASBlue}{ECMWF / GloFAS}{retrospective alignment and 28-day guidance.}
\SourceRow{NWSRust}{NOAA / NWS}{short-range guidance over common valid leads 1--8.}
\SourceRow{CovariateSage}{Transfer information}{precipitation, soil moisture, and the GDPC climate index.}
\end{SoftNote}

\SectionHead{Interpretation}{What this experiment supports}
\begin{PosterItems}
  \item The framework combines heterogeneous forecast sources while representing
  source discrepancy, lead-time structure, and distributional uncertainty.
